# The GRANT lines scripts/orch/hub-pg.sh start applies, and what the role reaches with them.
# Applied as postgres against database hub, on every start (hub-pg.sh:100-121).

$ scripts/orch/hub-pg.sh sql "SELECT rolname, rolsuper, rolcreatedb, rolcanlogin FROM pg_roles WHERE rolname IN ('hub','postgres') ORDER BY rolname"
 rolname  | rolsuper | rolcreatedb | rolcanlogin 
----------+----------+-------------+-------------
 hub      | f        | t           | t
 postgres | t        | t           | t
(2 rows)


$ scripts/orch/hub-pg.sh sql "SELECT has_parameter_privilege('hub','hub.writer','SET') AS set_hub_writer, has_schema_privilege('hub','public','CREATE') AS create_public, pg_has_role('hub','pg_monitor','MEMBER') AS pg_monitor"
 set_hub_writer | create_public | pg_monitor 
----------------+---------------+------------
 t              | t             | t
(1 row)


$ scripts/orch/hub-pg.sh sql "SELECT has_function_privilege('hub', fn, 'EXECUTE') AS exec FROM unnest(ARRAY['pg_control_system()','pg_control_checkpoint()','pg_control_init()','pg_control_recovery()','pg_create_restore_point(text)','pg_switch_wal()']) AS fn"
              fn               | exec 
-------------------------------+------
 pg_control_system()           | t
 pg_control_checkpoint()       | t
 pg_control_init()             | t
 pg_control_recovery()         | t
 pg_create_restore_point(text) | f
 pg_switch_wal()               | f
(6 rows)
